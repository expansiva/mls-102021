/** An agent of the organization as the cm keeps it, read only: its @name, whether it is on, what it is, its audience and the name of the person responsible for it. */
export interface AgentOverview {
  name: string;
  isOn: boolean;
  definition?: string;
  audience?: string;
  readonly responsibleName?: string;
}

export interface Agent_overviewContracts {
  /**
   * Purpose: Loads, in one call, the read-only overview of the chosen agent: what it is, who it serves (its audience), whether it is on and who is responsible for it.
   * Input: agentId: the cm identifier of the chosen agent, taken from the page URL.
   * Processing: Reads the agent from the cm (name, isOn, definition, audience, responsible). Resolves the responsible person’s organization user record to their readable name (OrganizationUser.details.identification.name), so the screen can show a name and not a technical id (noTechnicalIdsOnScreen). Returns only the fields that the grants allow organization members and company admins to see. Writes nothing.
   * Output: agent: the agent’s @name, whether it is on, its plain-language description, its audience (member, room or external) and the responsible person’s name (absent when no responsible person is set).
   */
  'agents.agent_overview.getAgentOverview': {
    kind: 'qry';
    input: { agentId: string };
    output: { agent: AgentOverview };
    rules: ['noTechnicalIdsOnScreen'];
    access: { actors: ['companyAdmin', 'organizationMember']; grants: ['adminManageAgents', 'adminChoosePeople', 'memberSeeAgents', 'memberSeePeopleNames']; scope: 'organization' };
  };
}
