import type { ExternalSystemAdapter, ExternalAdapterContext } from '/_102034_/l2/shared/contracts/connections/externalSystem.js';
import type { ListarConteudoInput, ListarConteudoItem, ListarConteudoOutput, LerArquivoInput, LerArquivoOutput, GravarArquivoInput, GravarArquivoOutput } from '/_102047_/l1/repoHost/layer_1_external/connections/github.js';

const headers = {
  Accept: 'application/vnd.github+json',
  Authorization: 'Bearer {{secret:chaveAcesso}}',
  'X-GitHub-Api-Version': '2022-11-28',
};

function contentsPath(repository: string, path: string): string {
  const repo = repository.trim().split('/').map(encodeURIComponent).join('/');
  const relative = path.replace(/^\/+/, '');
  const suffix = relative ? '/' + relative.split('/').map(encodeURIComponent).join('/') : '';
  return '/repos/' + repo + '/contents' + suffix;
}

function parseJson(body: string): unknown {
  try {
    return JSON.parse(body) as unknown;
  } catch {
    return undefined;
  }
}

function isObject(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function isSize(value: unknown): value is number {
  return typeof value === 'number' && Number.isSafeInteger(value) && value >= 0;
}

// GitHub's contents API requires the file bytes encoded as base64.
function encodeBase64Utf8(text: string): string {
  const bytes: number[] = [];
  for (let i = 0; i < text.length; i++) {
    let cp = text.codePointAt(i)!;
    if (cp > 0xffff) i++;
    else if (cp >= 0xd800 && cp <= 0xdfff) cp = 0xfffd;
    if (cp <= 0x7f) bytes.push(cp);
    else if (cp <= 0x7ff) bytes.push(0xc0 | (cp >> 6), 0x80 | (cp & 0x3f));
    else if (cp <= 0xffff) bytes.push(0xe0 | (cp >> 12), 0x80 | ((cp >> 6) & 0x3f), 0x80 | (cp & 0x3f));
    else bytes.push(0xf0 | (cp >> 18), 0x80 | ((cp >> 12) & 0x3f), 0x80 | ((cp >> 6) & 0x3f), 0x80 | (cp & 0x3f));
  }
  const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
  let result = '';
  for (let i = 0; i < bytes.length; i += 3) {
    const a = bytes[i];
    const b = i + 1 < bytes.length ? bytes[i + 1] : 0;
    const c = i + 2 < bytes.length ? bytes[i + 2] : 0;
    result += alphabet[a >> 2] + alphabet[((a & 3) << 4) | (b >> 4)] +
      (i + 1 < bytes.length ? alphabet[((b & 15) << 2) | (c >> 6)] : '=') +
      (i + 2 < bytes.length ? alphabet[c & 63] : '=');
  }
  return result;
}

const adapter: ExternalSystemAdapter = {
  async listarConteudo(input: unknown, ctx: ExternalAdapterContext) {
    const args = input as ListarConteudoInput;
    const repository = ctx.setting('repositorio');
    if (!repository || !repository.trim()) {
      return { ok: false, code: 'disabled', message: 'repositorio is not configured', missing: ['repositorio'] };
    }
    const response = await ctx.send('listarConteudo', {
      method: 'GET',
      path: contentsPath(repository, args.caminho),
      query: args.ramo !== undefined ? { ref: args.ramo } : undefined,
      headers,
    });
    if (!response.ok) return response;
    const body = parseJson(response.output.body);
    if (!Array.isArray(body)) {
      return { ok: false, code: 'invalid', message: 'GitHub directory response is not an array.' };
    }
    const itens: ListarConteudoItem[] = [];
    for (const entry of body as unknown[]) {
      if (!isObject(entry) || typeof entry.name !== 'string' || typeof entry.path !== 'string' ||
          (entry.type !== 'file' && entry.type !== 'dir') ||
          (entry.size !== undefined && !isSize(entry.size)) ||
          (entry.sha !== undefined && typeof entry.sha !== 'string')) {
        return { ok: false, code: 'invalid', message: 'GitHub directory response contains an invalid or unsupported item.' };
      }
      const item: ListarConteudoItem = {
        nome: entry.name,
        caminho: entry.path,
        tipo: entry.type === 'file' ? 'arquivo' : 'pasta',
      };
      if (typeof entry.size === 'number') item.tamanho = entry.size;
      if (typeof entry.sha === 'string') item.identificador = entry.sha;
      itens.push(item);
    }
    const output: ListarConteudoOutput = { itens };
    return { ok: true, output };
  },

  async lerArquivo(input: unknown, ctx: ExternalAdapterContext) {
    const args = input as LerArquivoInput;
    const repository = ctx.setting('repositorio');
    if (!repository || !repository.trim()) {
      return { ok: false, code: 'disabled', message: 'repositorio is not configured', missing: ['repositorio'] };
    }
    const response = await ctx.send('lerArquivo', {
      method: 'GET',
      path: contentsPath(repository, args.caminho),
      query: args.ramo !== undefined ? { ref: args.ramo } : undefined,
      headers,
    });
    if (!response.ok) return response;
    const body = parseJson(response.output.body);
    if (!isObject(body) || body.type !== 'file' || typeof body.path !== 'string' ||
        typeof body.name !== 'string' || typeof body.content !== 'string' ||
        (body.encoding !== 'base64' && body.encoding !== 'utf-8') ||
        !isSize(body.size) || typeof body.sha !== 'string') {
      return { ok: false, code: 'invalid', message: 'GitHub file response is invalid or has an unsupported encoding.' };
    }
    const output: LerArquivoOutput = {
      caminho: body.path,
      nome: body.name,
      conteudo: body.content,
      codificacao: body.encoding,
      tamanho: body.size,
      identificador: body.sha,
    };
    return { ok: true, output };
  },

  async gravarArquivo(input: unknown, ctx: ExternalAdapterContext) {
    const args = input as GravarArquivoInput;
    const repository = ctx.setting('repositorio');
    if (!repository || !repository.trim()) {
      return { ok: false, code: 'disabled', message: 'repositorio is not configured', missing: ['repositorio'] };
    }
    const payload: { message: string; content: string; branch?: string; sha?: string } = {
      message: args.mensagemRegistro,
      content: encodeBase64Utf8(args.conteudo),
    };
    if (args.ramo !== undefined) payload.branch = args.ramo;
    if (args.identificadorAnterior !== undefined) payload.sha = args.identificadorAnterior;
    const response = await ctx.send('gravarArquivo', {
      method: 'PUT',
      path: contentsPath(repository, args.caminho),
      headers: { ...headers, 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });
    if (!response.ok) return response;
    const body = parseJson(response.output.body);
    if (!isObject(body) || !isObject(body.content) || !isObject(body.commit) ||
        typeof body.content.path !== 'string' || typeof body.content.sha !== 'string' ||
        typeof body.commit.sha !== 'string') {
      return { ok: false, code: 'invalid', message: 'GitHub write response is invalid.' };
    }
    const output: GravarArquivoOutput = {
      caminho: body.content.path,
      identificador: body.content.sha,
      registro: body.commit.sha,
    };
    return { ok: true, output };
  },
};
export default adapter;
