/** An agent draft that is waiting for the admin: either ready for review or still being prepared by the cm from the admin’s description. */
export interface DraftInboxItem {
  name: string;
  situation: 'preparing' | 'ready' | 'refused' | 'activated' | 'discarded';
  summary?: string;
  prompt: string;
  requestedAt?: string;
  requestedBy?: string;
  draftId: string;
}

/** A company agent as the cm keeps it: its @name, who it talks to, whether it is on and the readable name of the person responsible for it. */
export interface AgentCard {
  name: string;
  talksTo?: string;
  isOn: boolean;
  readonly responsibleName?: string;
  agentId: string;
}

/** An agent draft in the cm with its name, its situation and, when refused, the reason the cm gave. */
export interface DraftCard {
  name: string;
  situation: 'preparing' | 'ready' | 'refused' | 'activated' | 'discarded';
  refusalReason?: string;
  draftId: string;
}

/** One entry of the company’s virtual employees: either an agent or a draft, exactly one of the two filled. */
export interface AgentsCollectionEntry {
  agent?: AgentCard;
  draft?: DraftCard;
}

/** A request for a new agent and what happened to it: sent, ready with the suggested tools turned on, refused with the reason, activated or discarded. */
export interface RequestEvent {
  name: string;
  situation: 'preparing' | 'ready' | 'refused' | 'activated' | 'discarded';
  requestedAt?: string;
  refusalReason?: string;
  suggestedTools?: string[];
  draftId: string;
}

export interface AgentsContracts {
  /**
   * Purpose: Loads in one call everything the agents home shows when it opens: the drafts waiting for the admin, the first page of agents and drafts, and the first page of the request history.
   * Input: pageSize: how many entries the first page of the agents collection and of the request history bring.
   * Processing: Reads agents and drafts from the cm for the caller’s organization. inbox: drafts in situation ’ready’ (waiting for review) or ’preparing’ (requests still being prepared), newest request first, with summary, prompt, requested on and the readable name of the requester. agents: page 1 of agents and drafts side by side; agents carry @name, talksTo, isOn and the responsible person resolved to OrganizationUser.details.identification.name; drafts carry name and situation limited to preparing, ready or refused, with refusalReason filled only when refused (refusalReasonShownWhenRefused). requests: page 1 of drafts in any situation, newest first, with readable names of the suggested tools for ready drafts (agentDraftSuggestedToolsOnWhenReady: those tools are already on for it). For an organizationMember, grants cover only Agent: inbox and requests come back empty and agents contains only agent entries with the allowed fields. Never returns technical ids (noTechnicalIdsOnScreen). Each DraftInboxItem carries draftId to open its page; the screen does not show it. Each AgentCard carries agentId to open its page; the screen does not show it. Each DraftCard carries draftId to open its page; the screen does not show it. Each RequestEvent carries draftId to open its page; the screen does not show it.
   * Output: inbox: plain list of waiting drafts; agents: first page { items, page, pageSize, hasMore } of agent or draft entries; requests: first page { items, page, pageSize, hasMore } of request events.
   */
  'agents.agents.loadAgentsPage': {
    kind: 'qry';
    input: { page: number; pageSize: number };
    output: { inbox: DraftInboxItem[]; agents: { items: AgentsCollectionEntry[]; page: number; pageSize: number; hasMore: boolean }; requests: { items: RequestEvent[]; page: number; pageSize: number; hasMore: boolean } };
    rules: ['refusalReasonShownWhenRefused', 'agentDraftSuggestedToolsOnWhenReady', 'noTechnicalIdsOnScreen'];
    access: { actors: ['companyAdmin', 'organizationMember']; grants: ['adminManageAgents', 'adminChoosePeople', 'memberSeeAgents', 'memberSeePeopleNames']; scope: 'organization' };
  };
  /**
   * Purpose: Brings a further page of the company’s agents and drafts.
   * Input: page: the page number wanted; pageSize: entries per page.
   * Processing: Same composition and order as loadAgentsPage.agents: agents with @name, talksTo, isOn and responsible person’s readable name; drafts in preparing, ready or refused with refusalReason only when refused. Members receive only agent entries with their allowed fields. No technical ids. Each AgentCard carries agentId to open its page; the screen does not show it. Each DraftCard carries draftId to open its page; the screen does not show it.
   * Output: agents: the requested page { items, page, pageSize, hasMore }.
   */
  'agents.agents.listAgentsCollection': {
    kind: 'qry';
    input: { page: number; pageSize: number };
    output: { agents: { items: AgentsCollectionEntry[]; page: number; pageSize: number; hasMore: boolean } };
    rules: ['refusalReasonShownWhenRefused', 'noTechnicalIdsOnScreen'];
    access: { actors: ['companyAdmin', 'organizationMember']; grants: ['adminManageAgents', 'adminChoosePeople', 'memberSeeAgents', 'memberSeePeopleNames']; scope: 'organization' };
  };
  /**
   * Purpose: Brings a further page of the history of requests for new agents.
   * Input: page: the page number wanted; pageSize: events per page.
   * Processing: Drafts of the organization in any situation, newest request first; refusalReason only on refused drafts; suggested tools as readable names for ready drafts, which are already turned on for it. Company admin only by grant. Each RequestEvent carries draftId to open its page; the screen does not show it.
   * Output: requests: the requested page { items, page, pageSize, hasMore }.
   */
  'agents.agents.listRequestHistory': {
    kind: 'qry';
    input: { page: number; pageSize: number };
    output: { requests: { items: RequestEvent[]; page: number; pageSize: number; hasMore: boolean } };
    rules: ['refusalReasonShownWhenRefused', 'agentDraftSuggestedToolsOnWhenReady', 'noTechnicalIdsOnScreen'];
    access: { actors: ['companyAdmin', 'organizationMember']; grants: ['adminManageAgents', 'adminChoosePeople', 'memberSeeAgents', 'memberSeePeopleNames']; scope: 'organization' };
  };
  /**
   * Purpose: Sends the admin’s plain-words description of a new virtual employee to the cm, which prepares a draft or refuses it.
   * Input: prompt: what the admin wrote describing the new agent and its functions.
   * Processing: Only the company admin may call it. Creates the draft request in the cm with the prompt; requestedBy is the logged-in admin and requestedAt the moment it is sent. The draft starts in situation ’preparing’; the cm later reports it ready or refused. Each DraftInboxItem carries draftId to open its page; the screen does not show it. Each AgentCard carries agentId to open its page; the screen does not show it. Each DraftCard carries draftId to open its page; the screen does not show it. Each RequestEvent carries draftId to open its page; the screen does not show it.
   * Output: The new draft in its ’preparing’ situation in the three shapes it takes: request (history event), inboxItem (waiting draft with prompt, requester and date) and collectionEntry (draft card).
   */
  'agents.agents.sendNewAgentRequest': {
    kind: 'cmd';
    writes: 'AgentDraft.create';
    input: { prompt: string };
    output: { request: RequestEvent; inboxItem: DraftInboxItem; collectionEntry: AgentsCollectionEntry };
    rules: [];
    access: { actors: ['companyAdmin', 'organizationMember']; grants: ['adminManageAgents', 'adminChoosePeople', 'memberSeeAgents', 'memberSeePeopleNames']; scope: 'organization' };
  };
}
