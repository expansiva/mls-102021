export type ListContentsInput = {
  /** Path of the folder inside the repository. Empty for the root. */
  path: string;
};

export type RepositoryEntry = {
  name: string;
  path: string;
  type: 'file' | 'dir';
  sha: string;
  size: number;
};

export type ListContentsOutput = {
  entries: RepositoryEntry[];
};

export type ReadFileInput = {
  /** Path of the file inside the repository. */
  path: string;
};

export type ReadFileOutput = {
  path: string;
  name: string;
  sha: string;
  size: number;
  /** File content, encoded as stated in `encoding`. */
  content: string;
  encoding: 'base64' | 'utf-8';
};

export type WriteFileInput = {
  /** Path of the file inside the repository. */
  path: string;
  /** New content of the file, in plain text. */
  content: string;
  /** Log message of the change. */
  message: string;
  /** Sha of the current file, when the file already exists and is being changed. */
  sha?: string;
};

export type WriteFileOutput = {
  path: string;
  /** Sha of the file written. */
  sha: string;
  /** Sha of the commit created. */
  commitSha: string;
  commitMessage: string;
};

export type RepositoryChangedInput = {
  /** Repository the notice is about. */
  repository: { fullName: string };
  /** Branch reference changed, e.g. refs/heads/main. */
  ref: string;
  /** Sha of the head after the change. */
  after: string;
  commits: {
    id: string;
    message: string;
    author: { name: string; email?: string; username?: string };
    added: string[];
    modified: string[];
    removed: string[];
  }[];
  headCommit?: {
    id: string;
    message: string;
    author: { name: string; email?: string; username?: string };
  };
  pusher?: { name: string; email?: string };
};

export type RepositoryChangedOutput = {
  received: boolean;
};
