import type { RequestContext } from '/_102034_/l1/server/layer_2_controllers/contracts.js';
import type {
  PlatformReleaseCatalogChangedInput,
  PlatformReleaseCatalogChangedOutput,
} from '/_102034_/l1/server/layer_1_external/connections/catalogs/platform.js';
import { updateTool } from '/_102047_/l1/agents/layer_2_application/usecases/updateTool.js';
import type { UpdateToolInput } from '/_102047_/l1/agents/layer_2_application/usecases/updateTool.js';

export const requests: { 'agents.platform.releaseCatalogChanged': (input: PlatformReleaseCatalogChangedInput, ctx: RequestContext) => Promise<PlatformReleaseCatalogChangedOutput> } = {
  'agents.platform.releaseCatalogChanged': async function (input, ctx) {
    if (typeof input.releaseId !== 'string' || input.releaseId.trim() === '') {
      throw new Error('releaseCatalogChanged: releaseId is required');
    }
    if (typeof input.hash !== 'string' || input.hash.trim() === '') {
      throw new Error('releaseCatalogChanged: hash of the release ' + input.releaseId + ' is required');
    }
    if (!Array.isArray(input.catalogs)) {
      throw new Error('releaseCatalogChanged: catalogs of the release ' + input.releaseId + ' must be a list');
    }

    await ctx.data.moduleData.runInTransaction(async (tx) => {
      const bound: RequestContext = { ...ctx, data: { ...ctx.data, moduleData: tx } };
      const addedAt = new Date().toISOString();

      for (const catalog of input.catalogs) {
        if (catalog.schemaVersion !== 'assistant-catalog-v1') {
          throw new Error('releaseCatalogChanged: unsupported catalog schemaVersion for module ' + String(catalog.module));
        }
        const actions = [...catalog.commands, ...catalog.queries];
        for (const action of actions) {
          if (typeof action.route !== 'string' || action.route === '') {
            throw new Error('releaseCatalogChanged: an action of module ' + catalog.module + ' has no route');
          }
          const toolInput: UpdateToolInput = {
            id: action.route,
            version: 1,
            toolKey: action.route,
            origin: 'catalog',
            moduleName: action.module,
            subject: action.subject,
            name: action.title,
            status: action.assistant === 'included' ? 'active' : 'inactive',
            details: {
              moduleTitle: action.moduleTitle ?? catalog.title ?? catalog.module,
              subjectTitle: action.subjectTitle ?? action.subject,
              description: action.purpose ?? action.title,
              releaseLabel: input.releaseId,
              addedAt,
            },
          };
          await updateTool(toolInput, bound);
        }
      }
    });

    return {};
  },
};
